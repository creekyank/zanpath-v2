 下面是 现在的  smart_factory.py文件
------------------------------------
import os
import json
import re
import subprocess

# --- 配置路径 ---
BASE_DIR = r"E:\zanpath v2"
INPUT_FILE = r"E:\Wisdom\article.txt"
TEMP_DIR = os.path.join(BASE_DIR, "temp")

def run_cmd(cmd):
    print(f">> 执行: {cmd}")
    # 使用 subprocess.run 的 shell 模式，注意编码处理
    result = subprocess.run(cmd, shell=True, cwd=BASE_DIR, capture_output=True, text=True, encoding='utf-8')
    if result.returncode != 0:
        print(f"⚠️ 命令执行警告/失败: {result.stderr}")
    return result.returncode == 0

def process_smart_factory():
    if not os.path.exists(INPUT_FILE):
        print(f"❌ 未找到输入文件: {INPUT_FILE}")
        return

    print("🚀 开始读取并解析文章...")
    with open(INPUT_FILE, 'r', encoding='utf-8') as f:
        content = f.read()
        # 按横杠分割文章 (支持 10 个以上的横杠)
        raw_articles = re.split(r'-{10,}', content)

    for raw_block in raw_articles:
        raw_block = raw_block.strip()
        if not raw_block: continue
        
        lines = [l.strip() for l in raw_block.split('\n') if l.strip()]
        if len(lines) < 10: continue # 确保内容足够完整

        try:
            # 1. 提取元数据 (使用更稳健的搜索方式)
            title = re.search(r"Title:\s*(.*)", raw_block).group(1).strip()
            module = re.search(r"Module:\s*(.*)", raw_block).group(1).strip()
            slug = re.search(r"Slug:\s*(.*)", raw_block).group(1).strip()
            img_keywords = re.search(r"Keywords:\s*(.*)", raw_block).group(1).strip()

            # 2. 提取 JSON (匹配最外层的花括号，支持多行)
            # 逻辑：找第一个 { 到第一个 } 为 EN，第二个 { 到第二个 } 为 ES
            json_blocks = re.findall(r'(\{.*?\})', raw_block, re.DOTALL)
            if len(json_blocks) < 2:
                print(f"❌ {title} 缺失 SEO JSON 数据")
                continue
            seo_en = json_blocks[0]
            seo_es = json_blocks[1]

            # 3. 提取正文内容 (Markdown/HTML)
            # 通过排除法：找到两个 JSON 块之间的内容通常是 EN，JSON 之后的是 ES
            # 或者简单的按长度切分大的段落
            parts = [p.strip() for p in raw_block.split('\n\n') if len(p.strip()) > 300]
            if len(parts) < 2:
                print(f"❌ {title} 正文解析失败（段落过短）")
                continue
            article_en = parts[0]
            article_es = parts[1]

            print(f"\n📦 正在处理文章: [{title}]")
            print(f"   模块: {module} | Slug: {slug}")

            # 4. 写入临时文件供 Node.js 读取 (清理旧数据)
            os.makedirs(TEMP_DIR, exist_ok=True)
            files = {
                "article_en.txt": article_en,
                "article_es.txt": article_es,
                "seo_en.json": seo_en,
                "seo_es.json": seo_es
            }
            for filename, data in files.items():
                with open(os.path.join(TEMP_DIR, filename), "w", encoding="utf-8") as f:
                    f.write(data)

            # 5. 调用 Node 脚本组装 JSON 文章 (对应 buildArticle.js)
            # 参数顺序：title, slug, module
            run_cmd(f'node scripts/buildArticle.js "{title}" "{slug}" "{module}"')

            # 6. 下载图片 (对应 downloadImage.js)
            # 参数顺序：title, slug, module, keywords
            run_cmd(f'node scripts/downloadImage.js "{title}" "{slug}" "{module}" "{img_keywords}"')

            # 7. 更新 Sitemap (Image Sitemap)
            # 这里的路径需要根据你的实际路由 routingFolder 匹配
            routing_folder = module
            if module == 'bazi': routing_folder = 'life-path'
            elif module == 'fengshui': routing_folder = 'space'
            elif module == 'face': routing_folder = 'visual'
            
            sitemap_path = f"en/wisdom/{routing_folder}/{slug}"
            run_cmd(f'node -e "require(\'./scripts/updateImageSitemap\')(\'{routing_folder}\', \'{sitemap_path}\', \'{slug}-1.webp\', \'{title}\', \'{title} image\')"')

        except Exception as e:
            print(f"❌ 解析文章 [{title if 'title' in locals() else '未知'}] 出错: {e}")

    # --- 流程结束 ---
    print("\n" + "="*30)
    print("✅ 所有文章处理完成！")
    print("="*30)
    
    # 最后统一 Build
    # if run_cmd("npm run build"):
    #     print("🚀 Build 成功！正在提交代码...")
    #     run_cmd("git add . && git commit -m 'feat: 批量更新 AI 文章' && git push")
    # else:
    #     print("❌ Build 失败，请查看上方日志。")

if __name__ == "__main__":
    process_smart_factory()
